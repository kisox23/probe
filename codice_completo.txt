pub fn update_ship_view(percentage: i32) -> String {
    let pct = percentage / 10;
    let mut updated = String::new();
    updated.push('[');

    let re_bar = Regex::new(r"\[[^\]]{10}\]").unwrap();

    for _ in 1..=pct {
        updated.push('=');
    }

    for _ in 0..(10 - pct) {
        updated.push(' ');
    }

    updated.push(']');

    re_bar
        .replace_all(SHIP, regex::NoExpand(updated.as_str()))
        .into_owned()
}

use colored::Colorize;
use serde::{Deserialize, Serialize};
use std::fs;
use std::sync::LazyLock;

use crate::item::Item;
use crate::item::Util::Harvester;
use crate::models::GameState;

static RECIPES: LazyLock<Vec<Recipe>> = LazyLock::new(|| {
    let content = fs::read_to_string("meta/recipes.json").expect("UNABLE TO LOCATE RECIPE FILE");
    let mut recipes: Vec<Recipe> = serde_json::from_str(&content).expect("UNABLE TO DESERIALIZE");

    for r in &mut recipes {
        r.ingredients.sort();
    }

    recipes
});

#[derive(Serialize, Deserialize, PartialEq, Eq)]
pub struct Recipe {
    product: Item,
    ingredients: Vec<Item>,
    quantity: i32,
}

pub fn craft(gs: &mut GameState, mut ingredients: Vec<Item>) -> bool {
    ingredients.sort();

    let Some(found) = RECIPES.iter().find(|r| r.ingredients == ingredients) else {
        return false;
    };

    for item in &ingredients {
        if gs.inventory.get_qty(item) == 0 {
            println!("You don't have those ingredients.");
            return false;
        }
    }

    if found.product == Item::Uti(Harvester) && gs.inventory.get_qty(&Item::Uti(Harvester)) > 0 {
        println!("You can only have one Harvester in your inventory.");
        return false;
    }

    for item in &found.ingredients {
        match item {
            Item::Ele(_) | Item::Mat(_) => gs.inventory.update_item(item, -1),
            Item::Misc(_) | Item::Uti(_) => {}
        }
    }

    gs.inventory.update_item(&found.product, found.quantity);

    println!("You crafted a {}!", found.product.to_string().green());

    return true;
}

use colored::Colorize;
use rand::seq::IndexedRandom;

use crate::{
    ascii_art::{
        INHABITANT_STANDING, INHABITANT_TELLING, PLANET, PLANET_INHABITATED, clear_and_render,
    },

struct Trade {
    item_received: Item,
    item_given: Item,
    qty_received: i32,
    qty_given: i32,
}

pub fn talk(gs: &mut GameState, planet: &Planet) {
    if !planet.inhabitated {
        println!("Who are you talking to?");
        return;
    }

    clear_and_render(INHABITANT_STANDING);
    let name = word_generator::make_name(rand::random_range(2..4), true);
    println!("You met the inhabitant {}.", name.yellow());
    loop {
        println!("What do you have to say?");
        let sentence = input_command("? > ");
        if sentence.len() < 3 {
            println!("Can't you express yourself?");
            continue;
        }
        let intent = classify(sentence.as_str());
        match intent {
            Intent::Trade => {
                trade(gs);
                continue;
            }
            Intent::Ask => {
                ask(planet);
                continue;
            }
            Intent::Greet => {
                println!("Hello there, stranger. What do you need?");
                continue;
            }
            Intent::Bye => {
                break;
            }
        }
    }

    if planet.inhabitated {
        clear_and_render(PLANET_INHABITATED);
    } else {
        clear_and_render(PLANET);
    }
}

fn trade(gs: &mut GameState) {
    clear_and_render(INHABITANT_TELLING);

    let mut rng = rand::rng();
    let trade: Trade;
    let kind = rand::random_bool(0.5);

    if kind {
        let elements: Vec<&Element> = ELEMENTS.sample(&mut rng, 2).collect();
        trade = Trade {
            item_received: Item::Ele(elements[0].clone()),
            item_given: Item::Ele(elements[1].clone()),
            qty_received: rand::random_range(1..3),
            qty_given: rand::random_range(1..3),
        };
    } else {
        let materials: Vec<&Material> = MATERIALS.sample(&mut rng, 2).collect();
        trade = Trade {
            item_received: Item::Mat(materials[0].clone()),
            item_given: Item::Mat(materials[1].clone()),
            qty_received: rand::random_range(1..3),
            qty_given: rand::random_range(1..3),
        };
    }

    println!(
        "They ask you if you want to trade {} x{} for {} x{}",
        trade.item_given, trade.qty_given, trade.item_received, trade.qty_received
    );

    loop {
        println!("Do you accept it?");
        let act = input_command("? > ");

        match act.trim() {
            "y" | "yes" => {
                if gs.inventory.get_qty(&trade.item_given) < trade.qty_given {
                    println!("{}", "You don't have the goods requested.".red());
                    break;
                }
                gs.inventory
                    .update_item(&trade.item_given, -trade.qty_given);
                gs.inventory
                    .update_item(&trade.item_received, trade.qty_given);
                println!("{}", "You got a deal!".blue());
                break;
            }
            "n" | "no" => {
                return;
            }
            &_ => {
                println!("It's a yes or no question.");
            }
        }
    }
}

fn ask(planet: &Planet) {
    let mut result: Vec<String> = Vec::new();
    for (_i, elem) in planet.elements.iter().enumerate() {
        if *elem.1 >= 15.0 {
            result.push(elem.0.to_string());
        }
    }

    print!("I'm pretty sure you can find ");
    let cnt = result.len();
    for (i, x) in result.iter().enumerate() {
        if i == cnt - 1 {
            print!("{} ", x);
            continue;
        }
        print!("{}, ", x);
    }
    if result.is_empty() {
        print!("nothing ");
    }
    print!("here.\n");
}

use serde::{Deserialize, Serialize};

use crate::item::{Element, Item, Material, SpaceFinding, Util};

use std::collections::HashMap;
use std::fmt::Display;

#[derive(Debug, Serialize, Deserialize)]
pub struct Inventory {
    pub materials: HashMap<Material, i32>,
    pub elements: HashMap<Element, i32>,
    pub misc: HashMap<SpaceFinding, i32>,
    pub utils: HashMap<Util, i32>,
}

impl Inventory {
    pub fn update_item(&mut self, item: &Item, qty: i32) {
        match item {
            Item::Mat(m) => *self.materials.entry(m.clone()).or_insert(0) += qty,
            Item::Ele(e) => *self.elements.entry(e.clone()).or_insert(0) += qty,
            Item::Misc(i) => *self.misc.entry(i.clone()).or_insert(0) += qty,
            Item::Uti(u) => *self.utils.entry(u.clone()).or_insert(0) += qty,
        }
    }

    pub fn print_all_items(&self) {
        let all_items = self
            .utils
            .iter()
            .map(|(k, v)| (k as &dyn Display, v))
            .chain(self.misc.iter().map(|(k, v)| (k as &dyn Display, v)))
            .chain(self.materials.iter().map(|(k, v)| (k as &dyn Display, v)))
            .chain(self.elements.iter().map(|(k, v)| (k as &dyn Display, v)));

        let mut c = 0;
        for (_i, elem) in all_items.enumerate() {
            if c % 4 == 0 && c > 0 {
                println!("");
            }
            if *elem.1 == 0 {
                continue;
            }
            c += 1;
            print!("<{} x{}>    ", elem.0, elem.1);
        }
        print!("\n");
    }

    pub fn get_qty(&self, item: &Item) -> i32 {
        match item {
            Item::Ele(e) => *self.elements.get(e).unwrap_or(&0),
            Item::Mat(m) => *self.materials.get(m).unwrap_or(&0),
            Item::Uti(u) => *self.utils.get(u).unwrap_or(&0),
            Item::Misc(misc) => *self.misc.get(misc).unwrap_or(&0),
        }
    }
}

#[derive(Clone, Copy, Hash, PartialEq, Eq)]
pub enum Intent {
    Greet,
    Trade,
    Ask,
    Bye,
}

pub struct Classifier {
    frequencies: HashMap<(Intent, String), usize>,
    words_per_intent: HashMap<Intent, usize>,
    sentences_per_intent: HashMap<Intent, usize>,
    word_set: HashSet<String>,
}

impl Classifier {
    pub fn predict(&self, sentence: &str) -> Intent {
        let tokens = tokenize(sentence);

        let total_sentences = TRAINING_DATA.len() as f64;
        let vocab_size = self.word_set.len() as f64;

        let mut best_score = f64::NEG_INFINITY;
        let mut best_intent = Intent::Greet;

        for (&intent, &n_sentences) in &self.sentences_per_intent {
            // calculate prior
            let mut score = (n_sentences as f64 / total_sentences).ln();

            let intent_word_total = *self.words_per_intent.get(&intent).unwrap_or(&0) as f64;

            // calculate the likelihood
            for token in &tokens {
                let count = *self.frequencies.get(&(intent, token.clone())).unwrap_or(&0) as f64;
                let prob = (count + 1.0) / (intent_word_total + vocab_size);
                score += prob.ln();
            }

            if score > best_score {
                best_score = score;
                best_intent = intent;
            }
        }

        best_intent
    }
}

pub fn tokenize(text: &str) -> Vec<String> {
    let lower = text.to_lowercase();

    let mut tokens: Vec<String> = Vec::new();
    for element in lower.split_whitespace() {
        let token: String = element.chars().filter(|c| c.is_alphabetic()).collect();
        if !token.is_empty() {
            tokens.push(token);
        }
    }

    tokens
}

pub fn train() -> Classifier {
    let mut frequencies: HashMap<(Intent, String), usize> = HashMap::new();
    let mut words_per_intent: HashMap<Intent, usize> = HashMap::new();
    let mut sentences_per_intent: HashMap<Intent, usize> = HashMap::new();
    let mut word_set: HashSet<String> = HashSet::new();

    // count how many occurences per Intent
    for &(intent, sentence) in TRAINING_DATA {
        *sentences_per_intent.entry(intent).or_insert(0) += 1;
        for token in tokenize(sentence) {
            *words_per_intent.entry(intent).or_insert(0) += 1;
            *frequencies.entry((intent, token.clone())).or_insert(0) += 1;
            word_set.insert(token);
        }
    }

    Classifier {
        frequencies,
        words_per_intent,
        sentences_per_intent,
        word_set,
    }
}

                    let Ok(item) = Item::from_str(item_str) else {
                        println!("Not an item");
                        continue;
                    };

                    let qty = match words.next() {
                        Some(qty_str) => match qty_str.parse::<i32>() {
                            Ok(n) if n > 0 => n,
                            _ => {
                                println!("Invalid quantity.");
                                continue;
                            }
                        },
                        None => 1,
                    };

                    if gs.inventory.get_qty(&item) < qty {
                        println!("You want to lose more than you have?");

                    }

                    gs.inventory.update_item(&item, -qty);
                    println!("Dropped.");

                } else if let Some(item_string) = cmd.strip_prefix("use ") {

                            }

                            }
                        }
                    }

                    }
                    continue;
                }
                println!("Not a command.");
            }
        }
    }

pub fn generate(next_id: u64, name_gen: &NameGenerator) -> Planet {

fn generate_resources(temperature: i32) -> HashMap<Element, f32> {
    // function that associates a percentage of spawning to a material based on temperature
    let mut raw_weights: HashMap<Element, f32> = HashMap::new();

    let heat_factor = ((temperature + 200) as f32 / 400.0).clamp(0.0, 1.0);
    let cold_factor = 1.0 - heat_factor;

    // hot materials
    raw_weights.insert(
        Element::Iron,
        (10.0 + heat_factor * 60.0) * rand::random_range(0.7..1.3),
    );
    raw_weights.insert(
        Element::Silicon,
        (15.0 + heat_factor * 50.0) * rand::random_range(0.7..1.3),
    );
    raw_weights.insert(
        Element::Titanium,
        (5.0 + heat_factor * 25.0) * rand::random_range(0.5..1.5),
    );
    raw_weights.insert(
        Element::Uranium,
        (2.0 + heat_factor * 15.0) * rand::random_range(0.2..1.8),
    );

    // cold materials
    raw_weights.insert(
        Element::Hydrogen,
        (5.0 + cold_factor * 70.0) * rand::random_range(0.7..1.3),
    );
    raw_weights.insert(
        Element::Helium,
        (5.0 + cold_factor * 50.0) * rand::random_range(0.7..1.3),
    );
    raw_weights.insert(
        Element::Nitrogen,
        (10.0 + cold_factor * 30.0) * rand::random_range(0.5..1.5),
    );
    raw_weights.insert(
        Element::Argon,
        (2.0 + cold_factor * 15.0) * rand::random_range(0.2..1.8),
    );

    let total_weight: f32 = raw_weights.values().sum();

    let mut percentages: HashMap<Element, f32> = HashMap::new();
    for (elem, weight) in raw_weights {
        let pct = (weight / total_weight) * 100.0;
        percentages.insert(elem, pct);
    }

    percentages
}

fn generate_materials() -> Vec<Material> {
    let mut materials: Vec<Material> = Vec::new();

    for elem in Material::iter() {
        let prob = rand::random_bool(0.4);
        if prob {
            materials.push(elem);
        }
    }

    materials
}

        return;

    if gs.inventory.get_qty(&Item::Uti(Util::Harvester)) == 0 {
        println!("You need a Harvester.");
        return;
    }

    let mut rng = rand::rng();
    let entries: Vec<(&Element, &f32)> = planet.elements.iter().collect();

    if let Ok((loot, _)) = entries.choose_weighted(&mut rng, |&(_, pct)| *pct) {

    }

const NAMES: &[&str] = &[
    "IP", "LO", "AK", "FEI", "MA", "GO", "NO", "KI", "TEI", "UU", "UL", "ROI", "KE", "A", "DO",
    "FA", "FEL", "UT", "JA", "CON", "R", "EI",

pub fn make_name(n_syllables: usize, name: bool) -> String {
    let mut rng = rand::rng();
    let mut choices: Vec<&str> = Vec::new();

    if name {
        for _ in 0..n_syllables {
            if let Some(syllable) = NAMES.choose(&mut rng) {
                choices.push(syllable);
            }
        }
        return choices.concat();
    }

    for _ in 0..n_syllables {
        if let Some(syllable) = SYLLABLES.choose(&mut rng) {
            choices.push(syllable);
        }
    }

    if n_syllables >= 5 {
        let half = choices.len() / 2;

        let (first, last) = choices.split_at(half);

        let a = capital_letter(&first.concat());
        let b = capital_letter(&last.concat());

        format!("{} {}", a, b)
    } else {
        capital_letter(&choices.concat())
    }
}

fn capital_letter(word: &str) -> String {
    let mut chars = word.chars(); // UTF-8, not single bytes!
    match chars.next() {
        None => String::new(),
        Some(before) => before.to_uppercase().collect::<String>() + chars.as_str(),
    }
}

pub struct MarkovModel {
    transitions: HashMap<char, Vec<char>>,
    // given a two-char window, give me the third one
}

impl MarkovModel {
    pub fn train(corpus: &[&str]) -> Self {
        let mut transitions: HashMap<char, Vec<char>> = HashMap::new();

        for word in corpus {
            let padded = format!("^{}$", word.to_lowercase()); // add tags for initial-final letters 
            let chars: Vec<char> = padded.chars().collect(); // create iterable of chars for every word

            for window in chars.windows(2) {
                // take a window of 3 chars
                let state = window[0]; // use the first two as reference
                let next = window[1]; // use the last as info 

                transitions.entry(state).or_default().push(next); // store the data
            }
        }

        Self { transitions }
    }

    pub fn generate(&self, min_len: usize, max_len: usize) -> String {
        let mut rng = rand::rng();
        let mut word = String::new();
        let mut state = '^'; // initial state

        while let Some(choices) = self.transitions.get(&state) {
            let &next = choices.choose(&mut rng).unwrap();

            // if it chooses '$' word ends
            if next == '$' {
                // if it's not long enough given alternatives
                if word.chars().count() < min_len && choices.len() > 1 {
                    continue;
                }
                break;
            }

            word.push(next);
            state = next; // new state is previous with the letter chosen before 

            if word.chars().count() >= max_len {
                break;
            }
        }

        capital_letter(&word)
    }
}

pub struct NameGenerator {
    hot: MarkovModel,
    cold: MarkovModel,
    mild: MarkovModel,
}

impl NameGenerator {
    pub fn new() -> Self {
        Self {
            hot: MarkovModel::train(HOT_CORPUS),
            cold: MarkovModel::train(COLD_CORPUS),
            mild: MarkovModel::train(MILD_CORPUS),
        }
    }

    pub fn make_name(&self, climate_class: u8) -> String {
        let mut rng = rand::rng();
        let model = match climate_class {
            1 => &self.hot,
            2 => &self.cold,
            _ => &self.mild,
        };

        let base_name = model.generate(4, 7);

        match rand::random_range(0..10) {
            0..=2 => {
                let suffix = SUFFIXES.choose(&mut rng).unwrap();
                format!("{}-{}", base_name, suffix)
            }
            3..=5 => {
                let second = model.generate(4, 6);
                format!("{} {}", base_name, second)
            }
            6..=7 => {
                let letter = (rand::random_range(65..=90) as u8) as char;
                let code = rand::random_range(10..99);
                format!("{}-{} {}", letter, code, base_name)
            }
            _ => base_name,
        }
    }
}
